sistema_tienda.py 
"""
Módulo de lógica de negocio para la aplicación móvil Tienda Mass.
Distrito de Ate, Lima - Proyecto de Calidad de Software ISO/IEC 25010:2023.
"""

def autenticar_usuario(correo: str, password: str) -> bool:
    """
    Dimensión: Fiabilidad y Seguridad.
    Valida el formato del correo y credenciales de acceso registradas.
    """
    if not correo or "@" not in correo:
        return False
    # Simulación de credenciales de usuario registrado
    if correo == "cliente@mass.pe" and password == "Mass2026*":
        return True
    return False


def verificar_disponibilidad_stock(stock_actual: int, cantidad_solicitada: int) -> bool:
    """
    Dimensión: Adecuación Funcional (Exactitud de inventario).
    Verifica que la cantidad solicitada sea positiva y no exceda el stock físico.
    """
    if cantidad_solicitada <= 0:
        return False
    return stock_actual >= cantidad_solicitada


def calcular_total_carrito(items: list, tiene_cupon_descuento: bool = False) -> float:
    """
    Dimensión: Eficiencia de Desempeño (Comportamiento transaccional).
    Calcula el subtotal sumando precio * cantidad.
    Aplica 10% de descuento si el cupón está activo y redondea a 2 decimales.
    """
    if not items:
        return 0.0

    subtotal = sum(item["precio"] * item["cantidad"] for item in items)
    
    if tiene_cupon_descuento:
        subtotal *= 0.90  # Aplica 10% de descuento promocional Mass

    return round(subtotal, 2)


def validar_cobertura_distrito(distrito: str) -> bool:
    """
    Dimensión: Capacidad de Interacción (Prevención de errores al usuario).
    Valida si el distrito ingresado forma parte de la zona de reparto (Ate).
    """
    coberturas_activas = ["ATE", "ATE VITARTE"]
    return distrito.strip().upper() in coberturas_activas
test_sistema_tienda.py 
import pytest
from sistema_tienda import (
    autenticar_usuario,
    verificar_disponibilidad_stock,
    calcular_total_carrito,
    validar_cobertura_distrito
)

def test_autenticar_usuario():
    """Prueba 1: Verificar inicio de sesión exitoso y rechazo de credenciales incorrectas."""
    assert autenticar_usuario("cliente@mass.pe", "Mass2026*") is True
    assert autenticar_usuario("cliente@mass.pe", "clave_invalida") is False
    assert autenticar_usuario("correo_sin_arroba", "123456") is False


def test_verificar_disponibilidad_stock():
    """Prueba 2: Verificar validación de inventario ante pedidos válidos e insuficientes."""
    assert verificar_disponibilidad_stock(stock_actual=10, cantidad_solicitada=5) is True
    assert verificar_disponibilidad_stock(stock_actual=2, cantidad_solicitada=5) is False
    assert verificar_disponibilidad_stock(stock_actual=10, cantidad_solicitada=0) is False


def test_calcular_total_carrito():
    """Prueba 3: Verificar cálculo de total con y sin descuento promocional."""
    productos = [
        {"nombre": "Arroz 1kg", "precio": 4.50, "cantidad": 2},   # S/ 9.00
        {"nombre": "Aceite 1L", "precio": 7.00, "cantidad": 1}    # S/ 7.00 -> Total = 16.00
    ]
    # Caso sin descuento
    assert calcular_total_carrito(productos, tiene_cupon_descuento=False) == 16.00
    # Caso con 10% de descuento (16.00 * 0.90 = 14.40)
    assert calcular_total_carrito(productos, tiene_cupon_descuento=True) == 14.40
    # Caso carrito vacío
    assert calcular_total_carrito([]) == 0.0


def test_validar_cobertura_distrito():
    """Prueba 4: Verificar restricción geográfica de delivery para el distrito de Ate."""
    assert validar_cobertura_distrito("Ate") is True
    assert validar_cobertura_distrito("  ate vitarte  ") is True
    assert validar_cobertura_distrito("Miraflores") is False
